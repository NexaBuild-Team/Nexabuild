// prisma\schema\architecture.prisma
model ArchitectureCompany {
  id                    String                  @id @default(uuid())
  name                  String
  createdAt             DateTime                @default(now())
  avatarUrl             String
  awardsWon             Int?
  budgetRangeLabel      String
  city                  String
  clientSatisfactionPct Int?
  country               String                  @default("Sri Lanka")
  coverImageUrl         String
  description           String
  email                 String                  @unique
  locationLabel         String
  projectCount          Int
  rating                Float
  reviewCount           Int
  updatedAt             DateTime
  yearsExperience       Int
  CompanySpecialization CompanySpecialization[]
  FirmService           FirmService[]
  FirmTeamMember        FirmTeamMember[]
  FirmTestimonial       FirmTestimonial[]
  HouseDesign           HouseDesign[]
}

// prisma\schema\construction.prisma
model ConstructionCompany {

  createdAt DateTime @default(now())
}

// prisma\schema\introspected.prisma
model CompanySpecialization {
  id                  String              @id
  label               String
  companyId           String
  ArchitectureCompany ArchitectureCompany @relation(fields: [companyId], references: [id], onDelete: Cascade)

  @@index([companyId])
}

model ConstructionPhase {
  id             String             @id
  phase          String
  detail         String
  status         ConstructionStatus @default(PENDING)
  sortOrder      Int                @default(0)
  designDetailId String
  DesignDetail   DesignDetail       @relation(fields: [designDetailId], references: [id], onDelete: Cascade)

  @@index([designDetailId])
}

model DesignDetail {
  id                     String               @id
  designId               String               @unique
  breadcrumb             String[]
  architectName          String
  architectFirm          String
  locationLabel          String
  completionYear         Int
  status                 DesignStatus         @default(PLANNING)
  heroImageUrl           String
  threeDVisualizationUrl String?
  priceLkr               Int
  bedrooms               Int
  bathrooms              Int
  sqftArea               Int
  garageSpaces           Int                  @default(0)
  overview               String
  designStyle            String?
  structureType          String?
  interiorFinish         String?
  createdAt              DateTime             @default(now())
  updatedAt              DateTime
  ConstructionPhase      ConstructionPhase[]
  HouseDesign            HouseDesign          @relation(fields: [designId], references: [id], onDelete: Cascade)
  DesignFeature          DesignFeature[]
  DesignGalleryImage     DesignGalleryImage[]
  DesignReview           DesignReview?
  FloorPlan              FloorPlan[]
  RelatedProject         RelatedProject[]
}

model DesignFeature {
  id           String       @id

  label        String
  sortOrder    Int          @default(0)
  detailId     String
  DesignDetail DesignDetail @relation(fields: [detailId], references: [id], onDelete: Cascade)

}

model DesignGalleryImage {

  sortOrder    Int          @default(0)
  detailId     String
  DesignDetail DesignDetail @relation(fields: [detailId], references: [id], onDelete: Cascade)

model DesignReview {

model HouseDesign {
  id                  String              @id
  title               String
  style               String
  priceLkr            Int
  architectName       String
  imageUrl            String
  bedrooms            Int
  bathrooms           Int
  sqftArea            Int

  @@index([priceLkr])
  @@index([style])

model HouseDesignTag {

  @@index([designId])
